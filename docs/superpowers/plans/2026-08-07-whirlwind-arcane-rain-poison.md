## Available helpers (in `kinds.ts`, already imported/defined)

- `rung(def, index): string` — a ramp rung as `#rrggbb`; index clamped. `RAMP_BASE` is `2`, so `RAMP_BASE + 1` is bright and `RAMP_BASE + 2` is hot.
- `base(def): string` — shorthand for `rung(def, RAMP_BASE)`.
- `rungRgba(def, index, alpha): string` — same rung as `rgba(...)` with an explicit alpha.
- `IMPACT_TIMING` — the timing spread used by `KINDS` entries. Set pieces never read it (the registry's `SET_PIECES` table is authoritative for generated defs), but `KindConfig` requires it.

// ---- whirlwind ------------------------------------------------------------
// A funnel, narrow at the ground and widening upward, drawn as a stack of
// horizontal ellipse bands that sway. The BACK half of each band goes on
// `behind` and the FRONT half on `front`, which is what makes the funnel read
// as enclosing the target rather than as a decal pasted over it. That split is
// the whole reason this is a set piece and not a matrix kind.
//
// Distinct from the `vortex` matrix kind, which is three stroked arms in a
// square frame (drawVortex, unchanged).

/** Design px above the ground point. Guarded against TALL_FRAME.groundY. */
export const WHIRLWIND_HEIGHT = 168;
/** Half-width of the topmost (widest) band. Guarded against the frame width. */
export const WHIRLWIND_TOP_HALF_W = 40;
/** Peak horizontal sway of a band's centre; stacks on the width for extent. */
export const WHIRLWIND_SWAY = 6;

const WHIRLWIND_BASE_HALF_W = 10;
const WHIRLWIND_BANDS = 9;
/** Vertical squash of each band's ellipse — an isometric-looking ring. */
const WHIRLWIND_BAND_FLATTEN = 0.32;

/** Geometry of band `i`, shared by all three layers so they cannot drift.
 *  `f` runs 0 at the ground to 1 at the top. Phase comes from `i` directly,
 *  NOT from indexFrac — see the module note on ring-of-fire's aliasing. */
function whirlwindBand(
  progress: number,
  cx: number,
  cy: number,
  i: number,
): { x: number; y: number; rx: number; ry: number } {
  const f = i / (WHIRLWIND_BANDS - 1);
  const rx =
    WHIRLWIND_BASE_HALF_W + (WHIRLWIND_TOP_HALF_W - WHIRLWIND_BASE_HALF_W) * f;
  return {
    x: cx + Math.sin(progress * Math.PI * 2 + f * 3.1) * WHIRLWIND_SWAY * f,
    y: cy - WHIRLWIND_HEIGHT * f,
    rx,
    ry: rx * WHIRLWIND_BAND_FLATTEN,
  };
}

/** Strokes one half of every band. `nearHalf` picks the lower arc (0..PI in
 *  canvas radians, y growing downward) for the front layer, the upper arc for
 *  the behind layer. */
function whirlwindBands(
  ctx: CanvasRenderingContext2D,
  progress: number,
  cx: number,
  cy: number,
  nearHalf: boolean,
  colour: string,
  alphaScale: number,
): void {
  ctx.strokeStyle = colour;
  ctx.lineWidth = 2;
  for (let i = 0; i < WHIRLWIND_BANDS; i++) {
    const b = whirlwindBand(progress, cx, cy, i);
    // Upper bands are fainter, so the funnel dissolves into the air rather
    // than ending on a hard rim.
    ctx.globalAlpha = alphaScale * (1 - (i / (WHIRLWIND_BANDS - 1)) * 0.45);
    ctx.beginPath();
    ctx.ellipse(
      b.x,
      b.y,
      b.rx,
      b.ry,
      0,
      nearHalf ? 0 : Math.PI,
      nearHalf ? Math.PI : Math.PI * 2,
    );
    ctx.stroke();
  }
  ctx.lineWidth = 1;
  ctx.globalAlpha = 1;
}

function drawWhirlwindBehind(
  ctx: CanvasRenderingContext2D,
  progress: number,
  def: VfxDef,
  cx: number,
  cy: number,
): void {
  // Ground scuff first, so the funnel's far wall strokes over it.
  ctx.globalAlpha = 0.5;
  ctx.fillStyle = base(def);
  ctx.beginPath();
  ctx.ellipse(cx, cy, 26, 9, 0, 0, Math.PI * 2);
  ctx.fill();
  ctx.globalAlpha = 1;

  whirlwindBands(ctx, progress, cx, cy, false, base(def), 0.7);
}

function drawWhirlwindTint(
  ctx: CanvasRenderingContext2D,
  progress: number,
  def: VfxDef,
  cx: number,
  cy: number,
): void {
  // Additive swirl on the target. Low alpha: the tint layer composites `add`
  // regardless of def.additive, so anything heavy here blows out.
  whirlwindBands(ctx, progress, cx, cy, true, rung(def, RAMP_BASE + 1), 0.28);
}

function drawWhirlwindFront(
  ctx: CanvasRenderingContext2D,
  progress: number,
  def: VfxDef,
  cx: number,
  cy: number,
): void {
  whirlwindBands(ctx, progress, cx, cy, true, rung(def, RAMP_BASE + 1), 1);

  // Debris orbiting the funnel. Each mote's angle and height come from its
  // index arithmetically, so the six stay genuinely spread.
  const motes = 6;
  ctx.fillStyle = rung(def, RAMP_BASE + 2);
  for (let i = 0; i < motes; i++) {
    const f = (i * 2 + 1) / (motes * 2);
    const b = whirlwindBand(progress, cx, cy, f * (WHIRLWIND_BANDS - 1));
    const a = progress * Math.PI * 4 + i * ((Math.PI * 2) / motes);
    ctx.globalAlpha = 0.85;
    ctx.beginPath();
    ctx.arc(
      b.x + Math.cos(a) * b.rx,
      b.y + Math.sin(a) * b.ry,
      2,
      0,
      Math.PI * 2,
    );
    ctx.fill();
  }
  ctx.globalAlpha = 1;
}
```

// ---- arcane-rain ----------------------------------------------------------
// A barrage of motes falling from off the top of the tall frame and popping on
// contact. They are STAGGERED — the eye reads a barrage that way and a curtain
// if they fall in unison — which is also the difference from the `rain` matrix
// kind, twelve simultaneous streaks in a square frame (drawRain, unchanged).

const ARCANE_MOTES = 7;
const ARCANE_START_ABOVE = 172;
const ARCANE_SPREAD = 34;
/** Fraction of the effect spent launching motes; the last one starts here. */
const ARCANE_STAGGER = 0.45;

/** Where mote `i` is, and whether it has landed.
 *
 *  Both the horizontal offset and the launch delay come from `i` through plain
 *  arithmetic — deliberately NOT indexFrac, whose 2.399 constant aliased
 *  ring-of-fire's tongues into a narrow band while looking independent. */
function arcaneMote(
  progress: number,
  i: number,
): { dx: number; fall: number; landed: boolean } {
  // Alternating outward placement: 0, +1, -1, +2, -2, ... spread evenly.
  const rank = Math.ceil(i / 2);
  const side = i % 2 === 0 ? 1 : -1;
  const dx = (rank / Math.ceil((ARCANE_MOTES - 1) / 2)) * ARCANE_SPREAD * side;

  const delay = (i / (ARCANE_MOTES - 1)) * ARCANE_STAGGER;
  const span = 1 - delay;
  const t = span <= 0 ? 1 : Math.min(1, Math.max(0, (progress - delay) / span));
  // Accelerating fall — a linear descent reads as a lift, not a drop.
  return { dx, fall: t * t, landed: t >= 1 };
}

function drawArcaneRainBehind(
  ctx: CanvasRenderingContext2D,
  progress: number,
  def: VfxDef,
  cx: number,
  cy: number,
): void {
  // Ground splashes, one per landed mote. An EMPTY behind frame before the
  // first mote lands is CORRECT — there is nothing behind the character yet.
  // Do not "fix" it by painting on the early frames.
  ctx.fillStyle = base(def);
  for (let i = 0; i < ARCANE_MOTES; i++) {
    const m = arcaneMote(progress, i);
    if (!m.landed) continue;
    ctx.globalAlpha = 0.55;
    ctx.beginPath();
    ctx.ellipse(cx + m.dx, cy, 9, 3.5, 0, 0, Math.PI * 2);
    ctx.fill();
  }
  ctx.globalAlpha = 1;
}

function drawArcaneRainTint(
  ctx: CanvasRenderingContext2D,
  progress: number,
  def: VfxDef,
  cx: number,
  cy: number,
): void {
  // Faint bloom following each mote. Composited `add`, so keep it light.
  ctx.fillStyle = rung(def, RAMP_BASE + 1);
  for (let i = 0; i < ARCANE_MOTES; i++) {
    const m = arcaneMote(progress, i);
    if (m.landed) continue;
    ctx.globalAlpha = 0.3;
    ctx.beginPath();
    ctx.arc(
      cx + m.dx,
      cy - ARCANE_START_ABOVE * (1 - m.fall),
      5,
      0,
      Math.PI * 2,
    );
    ctx.fill();
  }
  ctx.globalAlpha = 1;
}

function drawArcaneRainFront(
  ctx: CanvasRenderingContext2D,
  progress: number,
  def: VfxDef,
  cx: number,
  cy: number,
): void {
  for (let i = 0; i < ARCANE_MOTES; i++) {
    const m = arcaneMote(progress, i);
    const x = cx + m.dx;

    if (m.landed) {
      // Pop: a brief bright flare at the point of contact.
      ctx.globalAlpha = 0.9;
      ctx.fillStyle = rung(def, RAMP_BASE + 2);
      ctx.beginPath();
      ctx.arc(x, cy, 4, 0, Math.PI * 2);
      ctx.fill();
      continue;
    }

    const y = cy - ARCANE_START_ABOVE * (1 - m.fall);
    // Motion streak behind the mote, so a 4-frame fall reads as travel.
    ctx.globalAlpha = 0.7;
    ctx.strokeStyle = rung(def, RAMP_BASE + 1);
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.moveTo(x, y - 14);
    ctx.lineTo(x, y);
    ctx.stroke();

    ctx.globalAlpha = 1;
    ctx.fillStyle = rung(def, RAMP_BASE + 2);
    ctx.beginPath();
    ctx.moveTo(x, y - 4);
    ctx.lineTo(x + 3, y);
    ctx.lineTo(x, y + 4);
    ctx.lineTo(x - 3, y);
    ctx.closePath();
    ctx.fill();
  }
  ctx.lineWidth = 1;
  ctx.globalAlpha = 1;
}
```

```ts
// ---- poison ---------------------------------------------------------------
// A small puff of miasma with bubbles rising out of it, body-centred on a
// square frame. Deliberately the SMALLEST set piece: 400ms, no cycles, because
// it is meant to fire on every DoT tick and must clear before the next one.
//
// The shape name matches the `poison` STATUS id in the server's pub data. They
// are separate namespaces and the collision is harmless — do not rename either
// to "fix" it.

const POISON_BUBBLES = 5;
const POISON_CLOUD_RX = 20;
const POISON_RISE = 26;
// Extent: the cloud's rx plus the outermost bubble's offset and radius,
// 20 + 12 + 3 = 35 of the square frame's <=62 design-px half-budget.

/** Bubble `i`'s offset and progress. Derived from the index arithmetically —
 *  not indexFrac, per the module note on ring-of-fire's aliasing. */
function poisonBubble(
  progress: number,
  i: number,
): { dx: number; rise: number; r: number } {
  const spread = ((i % 3) - 1) * 12;
  const stagger = (i / POISON_BUBBLES) * 0.35;
  const t = Math.min(1, Math.max(0, (progress - stagger) / (1 - stagger)));
  return {
    dx: spread + (i % 2 === 0 ? 3 : -3),
    rise: t,
    // Bubbles swell as they rise, then pop — radius collapses past t = 0.8.
    r: t > 0.8 ? 3 * (1 - (t - 0.8) * 5) : 1.5 + t * 1.9,
  };
}

function drawPoisonBehind(
  ctx: CanvasRenderingContext2D,
  progress: number,
  def: VfxDef,
  cx: number,
  cy: number,
): void {
  // Low haze pooling under the body, spreading and thinning.
  ctx.globalAlpha = 0.5 * (1 - progress * 0.6);
  ctx.fillStyle = base(def);
  ctx.beginPath();
  ctx.ellipse(
    cx,
    cy + 6,
    POISON_CLOUD_RX * (0.7 + progress * 0.5),
    7 * (0.7 + progress * 0.5),
    0,
    0,
    Math.PI * 2,
  );
  ctx.fill();
  ctx.globalAlpha = 1;
}

function drawPoisonTint(
  ctx: CanvasRenderingContext2D,
  progress: number,
  def: VfxDef,
  cx: number,
  cy: number,
): void {
  // Sickly wash over the target. Composited `add`, so keep it faint.
  ctx.globalAlpha = 0.22 * (1 - progress);
  ctx.fillStyle = rung(def, RAMP_BASE + 1);
  ctx.beginPath();
  ctx.ellipse(cx, cy, POISON_CLOUD_RX, POISON_CLOUD_RX * 0.75, 0, 0, Math.PI * 2);
  ctx.fill();
  ctx.globalAlpha = 1;
}

function drawPoisonFront(
  ctx: CanvasRenderingContext2D,
  progress: number,
  def: VfxDef,
  cx: number,
  cy: number,
): void {
  for (let i = 0; i < POISON_BUBBLES; i++) {
    const b = poisonBubble(progress, i);
    if (b.r <= 0) continue;
    ctx.globalAlpha = 0.9 * (1 - b.rise * 0.5);
    ctx.fillStyle = rung(def, RAMP_BASE + (b.rise > 0.6 ? 2 : 1));
    ctx.beginPath();
    ctx.arc(cx + b.dx, cy - b.rise * POISON_RISE, b.r, 0, Math.PI * 2);
    ctx.fill();
  }
  ctx.globalAlpha = 1;
}
```
